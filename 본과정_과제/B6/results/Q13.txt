==============================================================================
Q13. [UPDATE] 반납 예정일이 지났는데 아직 '대여 중'인 기록을 '연체(OVERDUE)'로 바꾸고 연체료(하루 300원) 계산
==============================================================================

[SQL 1/4]
-- (1) 변경 전 : 상태별 건수
SELECT status, COUNT(*) AS cnt
FROM rental
GROUP BY status
ORDER BY status;

[실행 결과]
+----------+-----+
| status   | cnt |
+----------+-----+
| OVERDUE  |   1 |
| RENTED   |  11 |
| RETURNED |  26 |
+----------+-----+
(3행)

[SQL 2/4]
-- (2) 변경 : 조건에 맞는 행만 수정한다.  ※ WHERE 를 빼먹으면 모든 행이 바뀐다!
-- [SQLite 전용] julianday() = 날짜를 숫자로 바꿔 날짜 차이(일수)를 계산
UPDATE rental
SET status   = 'OVERDUE',
    late_fee = CAST(julianday('2026-10-08') - julianday(due_date) AS INTEGER) * 300
WHERE status = 'RENTED'
  AND due_date < '2026-10-08';

[실행 결과] 3행이 변경되었습니다.

[SQL 3/4]
-- (3) 변경 후 : 상태별 건수
SELECT status, COUNT(*) AS cnt
FROM rental
GROUP BY status
ORDER BY status;

[실행 결과]
+----------+-----+
| status   | cnt |
+----------+-----+
| OVERDUE  |   4 |
| RENTED   |   8 |
| RETURNED |  26 |
+----------+-----+
(3행)

[SQL 4/4]
-- (4) 방금 연체로 바뀐 기록 확인
SELECT rental_id, member_id, book_id, due_date, status, late_fee
FROM rental
WHERE status = 'OVERDUE'
ORDER BY rental_id;

[실행 결과]

| rental_id | member_id | book_id | due_date   | status  | late_fee |
+-----------+-----------+---------+------------+---------+----------+
|        27 |         6 |      19 | 2026-09-26 | OVERDUE |     3600 |
|        28 |         4 |      14 | 2026-10-02 | OVERDUE |     1800 |
|        29 |        10 |      15 | 2026-10-04 | OVERDUE |     1200 |
|        30 |         5 |      13 | 2026-10-06 | OVERDUE |      600 |
+-----------+-----------+---------+------------+---------+----------+
(4행)
